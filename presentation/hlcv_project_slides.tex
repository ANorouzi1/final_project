\begin{frame}{What This Project Tries To Solve}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.57\textwidth}
      \begin{itemize}
        \item Segment agricultural fields.
        \item The hard part is not only foreground/background segmentation.
        \item Neighboring fields often touch, so a normal binary mask can merge multiple parcels into one blob.
        \item Our contribution focuses the model and evaluation on boundaries, not only region overlap.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \metricbox{Input}{Sentinel-2}{\texttt{window\_a} image}
      \metricbox{Main target}{field mask}{2-class segmentation}
      \metricbox{Extra signal}{SDF}{geometry around field borders}
    \end{column}
  \end{columns}
  % Speaker cue: This slide is the high-level framing. The audience should understand that the project is about preserving parcel structure, not just making the white mask larger or smaller.
\end{frame}

\begin{frame}{Why Parcel Boundaries Matter}
  \begin{itemize}
    \item The white mask only says field or background.
    \item The colored map shows the real goal: each parcel should stay separate.
    \item If neighboring parcels get joined, the field area can look right but the map is still wrong.
  \end{itemize}

        \item One output head: binary field-mask logits.
        \item Loss: BCE + Dice + optional total variation.

          \item Mean IoU: image-wise overlap.
          \item Boundary IoU: border agreement, threshold 0.6 and radius 2.
        \end{itemize}
      \end{block}

  \end{columns}

\begin{frame}{Loss: Make Boundary Pixels Count More}
  \begin{block}{Training objective}
    \[
      \mathcal{L} =
      \lambda_{\mathrm{bce}}\mathcal{L}_{\mathrm{BCE}}^{\mathrm{boundary}}
      + \lambda_{\mathrm{dice}}\mathcal{L}_{\mathrm{Dice}}
      + \lambda_{\mathrm{sdf}}\mathcal{L}_{\mathrm{SmoothL1}}
      + \lambda_{\mathrm{tv}}\mathcal{L}_{\mathrm{TV}}
    \]
  \end{block}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \[
        w(D) = 1 + \alpha \exp \left(-\frac{|D|}{\sigma}\right)
      \]
      \small
      Here $D$ is the target SDF. Pixels near field borders have $D \approx 0$, so they receive larger BCE weight.
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{tabularx}{\textwidth}{@{}lX@{}}
        \toprule
        Config value & Current main setting \\
        \midrule
        BCE / Dice weights & 1.0 / 1.0 \\
        Boundary weight $\alpha$ & 20.0 \\
        Boundary width $\sigma$ & 0.12 \\
        SDF SmoothL1 weight & 0.1 \\
        TV weight & 0.0 \\
        \bottomrule
      \end{tabularx}
    \end{column}
  \end{columns}
  % Speaker cue: The method is intuitive: if errors at borders matter more for the final map, the loss should also pay more attention there.
\end{frame}

\begin{frame}{Interpretation So Far}
  \begin{itemize}
    \item Boundary-focused training improves the metric most connected to the project goal: separating field borders.
    \item The current validation result improves Boundary IoU without lowering mIoU.
    \item The \config{d0} ablation is close to the SDF-loss run, so the final report should be careful: the boundary-weighted BCE seems to drive much of the current gain.
    \item The remaining sweeps should decide whether the SDF auxiliary head adds a consistent benefit beyond boundary BCE.
    \item Final claims should use the test split and a small qualitative panel.
  \end{itemize}
  % Speaker cue: This slide is where you sound honest. Do not overclaim that the SDF head alone caused everything until the final ablations are complete.
\end{frame}
